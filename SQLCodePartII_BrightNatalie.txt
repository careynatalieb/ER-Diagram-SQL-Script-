CREATE TABLE Member (
    MemberID INT PRIMARY KEY, 
    FirstName VARCHAR(15) NOT NULL,
    LastName VARCHAR(15) NOT NULL,
    DateOfBirth DATE,
    Address VARCHAR(30),
    PhoneNumber VARCHAR(11),
    DateJoined DATE
);

CREATE TABLE Volunteer (
    VolunteerID INT PRIMARY KEY,
    MemberID INT,
    FirstName VARCHAR(15) NOT NULL,
    LastName VARCHAR(15) NOT NULL,
    Availability VARCHAR(100),
    FOREIGN KEY (MemberID) REFERENCES Member(MemberID)
);

CREATE TABLE Event ( 
    EventID INT PRIMARY KEY,
    Name VARCHAR(50) NOT NULL,
    Date DATE,
    Time TIME,
    Location VARCHAR(100),
    Description VARCHAR(200),
    VolunteerID INT,
    FOREIGN KEY (VolunteerID) REFERENCES Volunteer(VolunteerID)
);

CREATE TABLE Donations (
    DonationID INT PRIMARY KEY,
    MemberID INT NOT NULL,
    Date DATE,
    Type VARCHAR(50),
    Amount DECIMAL(10,2),
    FOREIGN KEY (MemberID) REFERENCES Member(MemberID)
);

SELECT 
    d.DonationID,
    d.Date,
    d.Type,
    d.Amount,
    m.MemberID,
    CONCAT(m.FirstName, ' ', m.LastName) AS MemberName
FROM Donations d
INNER JOIN Member m ON d.MemberID = m.MemberID
ORDER BY d.DonationID;

SELECT 
    e.EventID,
    e.Name AS EventName,
    e.Date,
    e.Location,
    v.VolunteerID,
    CONCAT(v.FirstName, ' ', v.LastName) AS VolunteerName
FROM Event e
LEFT JOIN Volunteer v ON e.VolunteerID = v.VolunteerID
ORDER BY e.EventID;

SELECT 
    MemberID, 
    CONCAT(FirstName, ' ', LastName) AS MemberName
FROM Member
WHERE MemberID IN (
    SELECT MemberID 
    FROM Donations
    GROUP BY MemberID
    HAVING SUM(Amount) > (SELECT AVG(Amount) FROM Donations)
)
ORDER BY MemberID;

SELECT 
    EventID,
    Name AS EventName,
    Date,
    Location
FROM Event
WHERE Date IN (
    SELECT Date 
    FROM Event 
    WHERE YEAR(Date) = 2025
)
ORDER BY EventID;

SELECT 
    m.MemberID,
    CONCAT(m.FirstName, ' ', m.LastName) AS MemberName,
    SUM(d.Amount) AS TotalDonated
FROM Member m
LEFT JOIN Donations d ON m.MemberID = d.MemberID
GROUP BY m.MemberID, m.FirstName, m.LastName
ORDER BY TotalDonated DESC;

SELECT 
    COUNT(*) AS DonationCount,
    AVG(Amount) AS AverageDonation,
    SUM(Amount) AS TotalDonations
FROM Donations;

SELECT 
    MemberID,
    UPPER(CONCAT(FirstName, ' ', LastName)) AS NameUpper,
    ROUND(DATEDIFF(CURRENT_DATE, DateJoined) / 365, 1) AS YearsSinceJoined
FROM Member
ORDER BY MemberID;

SELECT 
    m.MemberID,
    CONCAT(m.FirstName, ' ', m.LastName) AS MemberName,
    COUNT(d.DonationID) AS NumberOfDonations
FROM Member m
JOIN Donations d ON m.MemberID = d.MemberID
GROUP BY m.MemberID, m.FirstName, m.LastName
ORDER BY NumberOfDonations DESC;
